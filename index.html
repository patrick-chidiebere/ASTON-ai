Cloudflare Worker Handler for Video & Images:

export default {
  async fetch(request, env) {
    const { prompt, type } = await request.json();
    const model = type === 'video' 
      ? '@cf/minimax/h3-max' 
      : '@cf/black-forest-labs/flux-1-schnell';
    const response = await env.AI.run(model, { prompt });
    return new Response(response, { headers: { 'Content-Type': 'video/mp4' } });
  }
};
